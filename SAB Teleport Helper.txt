local notificationToken = 0

local function showNotification(gearName)
	notificationToken += 1
	local token = notificationToken

	notificationText.Text = "Gear selected: " .. gearName
	notification.Visible = true

	notification.Position = UDim2.new(0.5, 0, 0, -48)

	TweenService:Create(
		notification,
		TweenInfo.new(0.25, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
		{Position = UDim2.new(0.5, 0, 0, 8)}
	):Play()

	task.delay(1.8, function()
		if token ~= notificationToken then
			return
		end

		local fade = TweenService:Create(
			notification,
			TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
			{
				Position = UDim2.new(0.5, 0, 0, -48),
				BackgroundTransparency = 1
			}
		)

		local textFade = TweenService:Create(
			notificationText,
			TweenInfo.new(0.3),
			{TextTransparency = 1}
		)

		local strokeFade = TweenService:Create(
			notificationStroke,
			TweenInfo.new(0.3),
			{Transparency = 1}
		)

		fade:Play()
		textFade:Play()
		strokeFade:Play()

		fade.Completed:Wait()

		if token == notificationToken then
			notification.Visible = false
			notification.BackgroundTransparency = 0.05
			notificationText.TextTransparency = 0
			notificationStroke.Transparency = 0
		end
	end)
end

--// Refresh Gears
local function refreshGears()
	for _, child in ipairs(list:GetChildren()) do
		if child:IsA("TextButton") then
			child:Destroy()
		end
	end

	local backpack = player:WaitForChild("Backpack")

	for _, tool in ipairs(backpack:GetChildren()) do
		if tool:IsA("Tool") then

			local button = Instance.new("TextButton")
			button.Size = UDim2.new(1, -6, 0, 28)
			button.Text = tool.Name
			button.TextColor3 = Color3.fromRGB(255, 255, 255)
			button.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
			button.BorderSizePixel = 0
			button.Font = Enum.Font.Gotham
			button.TextSize = 12
			button.Parent = list

			local buttonCorner = Instance.new("UICorner")
			buttonCorner.CornerRadius = UDim.new(0, 4)
			buttonCorner.Parent = button

			button.MouseButton1Click:Connect(function()
				local character = player.Character
				local humanoid = character and character:FindFirstChildOfClass("Humanoid")

				if humanoid then
					humanoid:EquipTool(tool)
				end

				dropdown.Text = tool.Name .. " ▼"
				list.Visible = false

				showNotification(tool.Name)
			end)
		end
	end
end

dropdown.MouseButton1Click:Connect(function()
	refreshGears()
	list.Visible = not list.Visible
end)

--// Position System
local savedCFrame

setPos.MouseButton1Click:Connect(function()
	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")

	if root then
		savedCFrame = root.CFrame

		setPos.Text = "Position Set!"

		task.delay(1, function()
			if setPos then
				setPos.Text = "Set Position"
			end
		end)
	end
end)

teleport.MouseButton1Click:Connect(function()
	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")

	if root and savedCFrame then
		root.CFrame = savedCFrame
	end
end)

--// Draggable Menu
local dragging = false
local dragStart
local startPos

title.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPos = main.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

UIS.InputChanged:Connect(function(input)
	if dragging and (
		input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch
	) then

		local delta = input.Position - dragStart

		main.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)

--// Black → White → Black Menu Animation

	while main.Parent do

		local toWhite = TweenService:Create(
			main,
			TweenInfo.new(2.5, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut),
			{BackgroundColor3 = Color3.fromRGB(255, 255, 255)}
		)

		toWhite:Play()
		toWhite.Completed:Wait()

		local toBlack = TweenService:Create(
			main,
			TweenInfo.new(2.5, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut),
			{BackgroundColor3 = Color3.fromRGB(0, 0, 0)}
		)

		toBlack:Play()
		toBlack.Completed:Wait()
	end
end)
